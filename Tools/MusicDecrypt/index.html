function addFiles(list){ if(isDecrypting) return; let added=0, rejected=[]; for(const f of list){ const fmt=detectFormat(f.name); if(!fmt){ // 不支持的格式：不静默丢弃，记录原因后加入列表（标记为 err） rejected.push(f); continue; } files.push({file:f,name:f.name,size:f.size,status:'pending',blob:null,url:null,err:null,fmt}); added++; } // 被拒绝的文件也加入列表，展示原因，避免「选了文件却毫无反应」 for(const f of rejected){ const why = explainUnsupported(f.name); files.push({file:f,name:f.name,size:f.size,status:'err',blob:null,url:null,err:why,fmt:'unsupported',unsupported:true}); console.warn('[addFiles] 不支持的文件:', f.name, '→', why); } if(rejected.length){ errMsg('已跳过 '+rejected.length+' 个不支持的文件：'+rejected.map(f=>f.name).join('、') +'（支持 .kgm/.ncm/.qmc/.mflac/.mgg/.kwm/.xm/.tm 等，详见页底说明）'); } renderList(); go.disabled=files.filter(f=>!f.unsupported).length===0; ba.classList.remove('show'); if(!rejected.length) er.classList.remove('show'); } // 给出更具体的拒绝原因，帮助用户判断 function explainUnsupported(name){ const n=(name||'').toLowerCase(); if(/\.kgg(\.(flac|mp3))?$/.test(n)) return '暂不支持 .kgg（酷狗新格式，与 .kgm 算法不同）'; if(/\.mp3$|\.flac$|\.m4a$|\.wav$|\.ogg$|\.aac$/.test(n)) return '这不是加密文件，无需解密'; const ext=(n.match(/\.[^.\/]+$/)||[''])[0]; return '不支持的格式'+(ext?'：'+ext:''); }
